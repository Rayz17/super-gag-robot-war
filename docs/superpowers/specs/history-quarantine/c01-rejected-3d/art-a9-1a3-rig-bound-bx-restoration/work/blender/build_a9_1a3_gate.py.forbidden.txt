def color(hex_value: str) -> tuple[float, float, float, float]:
    value = hex_value.lstrip("#")
    return tuple(int(value[index : index + 2], 16) / 255 for index in (0, 2, 4)) + (1.0,)


def darker(rgba: tuple[float, float, float, float], factor: float) -> tuple[float, float, float, float]:
    return tuple(channel * factor for channel in rgba[:3]) + (rgba[3],)


def cel_material(name: str, hex_value: str) -> bpy.types.Material:
    material = bpy.data.materials.new(name)
    material.use_nodes = True
    nodes = material.node_tree.nodes
    links = material.node_tree.links
    nodes.clear()
    output = nodes.new("ShaderNodeOutputMaterial")
    diffuse = nodes.new("ShaderNodeBsdfDiffuse")
    shader_to_rgb = nodes.new("ShaderNodeShaderToRGB")
    ramp = nodes.new("ShaderNodeValToRGB")
    emission = nodes.new("ShaderNodeEmission")
    base = color(hex_value)
    diffuse.inputs["Color"].default_value = base
    diffuse.inputs["Roughness"].default_value = 0.78
    ramp.color_ramp.interpolation = "CONSTANT"
    ramp.color_ramp.elements.remove(ramp.color_ramp.elements[1])
    first = ramp.color_ramp.elements[0]
    first.position = 0.28
    first.color = darker(base, 0.28)
    middle = ramp.color_ramp.elements.new(0.50)
    middle.color = darker(base, 0.60)
    bright = ramp.color_ramp.elements.new(0.72)
    bright.color = base
    emission.inputs["Strength"].default_value = 1.0
    links.new(diffuse.outputs["BSDF"], shader_to_rgb.inputs["Shader"])
    links.new(shader_to_rgb.outputs["Color"], ramp.inputs["Fac"])
    links.new(ramp.outputs["Color"], emission.inputs["Color"])
    links.new(emission.outputs["Emission"], output.inputs["Surface"])
    return material


def reset_scene() -> None:
    bpy.ops.object.select_all(action="SELECT")
    bpy.ops.object.delete(use_global=False)
    for collection in tuple(bpy.data.collections):
        if collection.name != "Collection":
            bpy.data.collections.remove(collection)


def new_collection(name: str) -> bpy.types.Collection:
    collection = bpy.data.collections.new(name)
    bpy.context.scene.collection.children.link(collection)
    return collection


def relink(obj: bpy.types.Object, collection: bpy.types.Collection) -> None:
    for existing in tuple(obj.users_collection):
        existing.objects.unlink(obj)
    collection.objects.link(obj)


def component_empty(name: str, collection: bpy.types.Collection, root: bpy.types.Object) -> bpy.types.Object:
    empty = bpy.data.objects.new(name, None)
    empty.empty_display_type = "PLAIN_AXES"
    empty.parent = root
    collection.objects.link(empty)
    return empty


def bevelled_cube(
    name: str,
    size: tuple[float, float, float],
    location: tuple[float, float, float],
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
    rotation: tuple[float, float, float] = (0, 0, 0),
    bevel: float = 0.08,
) -> bpy.types.Object:
    bpy.ops.mesh.primitive_cube_add(location=location, rotation=rotation)
    obj = bpy.context.object
    obj.name = name
    obj.scale = (size[0] / 2, size[1] / 2, size[2] / 2)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    obj.data.materials.append(material)
    modifier = obj.modifiers.new("HARD_EDGE_CHAMFER", "BEVEL")
    modifier.width = min(bevel, min(size) * 0.2)
    modifier.segments = 1
    modifier.affect = "EDGES"
    obj.parent = parent
    relink(obj, collection)
    return obj


def cylinder(
    name: str,
    radius: float,
    depth: float,
    location: tuple[float, float, float],
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
    rotation: tuple[float, float, float] = (math.pi / 2, 0, 0),
    vertices: int = 12,
) -> bpy.types.Object:
    bpy.ops.mesh.primitive_cylinder_add(vertices=vertices, radius=radius, depth=depth, location=location, rotation=rotation)
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(material)
    modifier = obj.modifiers.new("RIM_CHAMFER", "BEVEL")
    modifier.width = min(0.045, depth * 0.12)
    modifier.segments = 1
    obj.parent = parent
    relink(obj, collection)
    return obj


def ico(
    name: str,
    location: tuple[float, float, float],
    scale: tuple[float, float, float],
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
    rotation: tuple[float, float, float] = (0, 0, 0),
) -> bpy.types.Object:
    bpy.ops.mesh.primitive_ico_sphere_add(subdivisions=2, radius=1, location=location, rotation=rotation)
    obj = bpy.context.object
    obj.name = name
    obj.scale = scale
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    obj.data.materials.append(material)
    obj.parent = parent
    relink(obj, collection)
    return obj


def panel_prism(
    name: str,
    points_xz: tuple[tuple[float, float], ...],
    y_front: float,
    y_back: float,
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
    bevel: float = 0.05,
) -> bpy.types.Object:
    """Create a bevelled armor plate from a front-view polygon with real depth."""
    count = len(points_xz)
    vertices = [(x, y_front, z) for x, z in points_xz] + [(x, y_back, z) for x, z in points_xz]
    faces = [tuple(range(count)), tuple(range(count, count * 2))]
    for index in range(count):
        nxt = (index + 1) % count
        faces.append((index, nxt, count + nxt, count + index))
    mesh = bpy.data.meshes.new(f"{name}_MESH")
    mesh.from_pydata(vertices, [], faces)
    mesh.update()
    obj = bpy.data.objects.new(name, mesh)
    obj.data.materials.append(material)
    obj.parent = parent
    collection.objects.link(obj)
    modifier = obj.modifiers.new("ARMOR_EDGE_CHAMFER", "BEVEL")
    modifier.width = bevel
    modifier.segments = 1
    modifier.affect = "EDGES"
    return obj


def torus(
    name: str,
    major_radius: float,
    minor_radius: float,
    location: tuple[float, float, float],
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
    rotation: tuple[float, float, float] = (math.pi / 2, 0, 0),
) -> bpy.types.Object:
    bpy.ops.mesh.primitive_torus_add(
        align="WORLD",
        major_segments=24,
        minor_segments=8,
        location=location,
        rotation=rotation,
        major_radius=major_radius,
        minor_radius=minor_radius,
    )
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(material)
    obj.parent = parent
    relink(obj, collection)
    return obj


def segmented_fist(
    name: str,
    location: tuple[float, float, float],
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
) -> None:
    x, y, z = location
    bevelled_cube(f"{name}_palm", (0.62, 0.52, 0.44), (x, y, z), material, parent, collection, bevel=0.08)
    for index, offset in enumerate((-0.23, -0.075, 0.075, 0.23)):
        bevelled_cube(
            f"{name}_finger_{index}",
            (0.13, 0.5, 0.18),
            (x + offset, y - 0.06, z - 0.24),
            material,
            parent,
            collection,
            rotation=(0.08, 0, 0),
            bevel=0.035,
        )


def beam(
    name: str,
    start: tuple[float, float, float],
    end: tuple[float, float, float],
    width: float,
    depth: float,
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
    bevel: float = 0.06,
) -> bpy.types.Object:
    start_v = Vector(start)
    end_v = Vector(end)
    direction = end_v - start_v
    midpoint = (start_v + end_v) / 2
    obj = bevelled_cube(name, (width, depth, direction.length), midpoint, material, parent, collection, bevel=bevel)
    obj.rotation_mode = "QUATERNION"
    obj.rotation_quaternion = Vector((0, 0, 1)).rotation_difference(direction.normalized())
    return obj


def tapered_strut(
    name: str,
    start: tuple[float, float, float],
    end: tuple[float, float, float],
    radius_start: float,
    radius_end: float,
    material: bpy.types.Material,
    parent: bpy.types.Object,
    collection: bpy.types.Collection,
    vertices: int = 7,
) -> bpy.types.Object:
    """Faceted non-industrial load strut aligned to two rig points."""
    start_v = Vector(start)
    end_v = Vector(end)
    direction = end_v - start_v
    midpoint = (start_v + end_v) / 2
    bpy.ops.mesh.primitive_cone_add(
        vertices=vertices,
        radius1=radius_end,
        radius2=radius_start,
        depth=direction.length,
        location=midpoint,
    )
    obj = bpy.context.object
    obj.name = name
    obj.rotation_mode = "QUATERNION"
    obj.rotation_quaternion = Vector((0, 0, 1)).rotation_difference(direction.normalized())
    obj.data.materials.append(material)
    obj.parent = parent
    relink(obj, collection)
    bevel = obj.modifiers.new("ORGANIC_EDGE_BREAK", "BEVEL")
    bevel.width = 0.035
    bevel.segments = 1
    return obj


def add_keypoint(name: str, location: tuple[float, float, float], root: bpy.types.Object, collection: bpy.types.Collection) -> None:
    empty = bpy.data.objects.new(f"KP_{name}", None)
    empty.location = location
    empty.empty_display_type = "SPHERE"
    empty.empty_display_size = 0.08
    empty.hide_render = True
    empty.parent = root
    collection.objects.link(empty)

def aim(camera: bpy.types.Object, target: tuple[float, float, float]) -> None:
    direction = Vector(target) - camera.location
    camera.rotation_euler = direction.to_track_quat("-Z", "Y").to_euler()


def set_collection_visibility(collections: tuple[bpy.types.Collection, ...], visible: bpy.types.Collection) -> None:
    for collection in collections:
        collection.hide_render = collection != visible
        collection.hide_viewport = collection != visible


def export_keypoints(root: bpy.types.Object, camera: bpy.types.Object, path: Path) -> None:
    scene = bpy.context.scene
    data = {}
    for obj in root.children_recursive:
        if not obj.name.startswith("KP_"):
            continue
        coord = bpy_extras.object_utils.world_to_camera_view(scene, camera, obj.matrix_world.translation)
        data[obj.name[3:]] = {"x": round(coord.x * 1024, 3), "y": round((1 - coord.y) * 1024, 3), "depth": round(coord.z, 6)}
    path.write_text(json.dumps(data, indent=2, sort_keys=True) + "\n", encoding="utf-8")


def render_view(
    collection: bpy.types.Collection,
    all_collections: tuple[bpy.types.Collection, ...],
    root: bpy.types.Object,
    camera: bpy.types.Object,
    camera_location: tuple[float, float, float],
    target: tuple[float, float, float],
    ortho_scale: float,
    filename: str,
) -> None:
    set_collection_visibility(all_collections, collection)
    camera.location = camera_location
    camera.data.ortho_scale = ortho_scale
    aim(camera, target)
    bpy.context.scene.render.filepath = str(OUT / filename)
    bpy.ops.render.render(write_still=True)
    export_keypoints(root, camera, OUT / filename.replace(".png", "_keypoints.json"))
