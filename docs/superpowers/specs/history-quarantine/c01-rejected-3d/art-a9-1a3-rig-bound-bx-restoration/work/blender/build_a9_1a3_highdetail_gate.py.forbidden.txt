def rgba(value: str) -> tuple[float, float, float, float]:
    value = value.lstrip("#")
    return tuple(int(value[i:i + 2], 16) / 255.0 for i in (0, 2, 4)) + (1.0,)


def shade(c: tuple[float, float, float, float], f: float) -> tuple[float, float, float, float]:
    return tuple(max(0.0, min(1.0, x * f)) for x in c[:3]) + (1.0,)


def cel_mat(name: str, value: str) -> bpy.types.Material:
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()
    out = nodes.new("ShaderNodeOutputMaterial")
    diff = nodes.new("ShaderNodeBsdfDiffuse")
    to_rgb = nodes.new("ShaderNodeShaderToRGB")
    ramp = nodes.new("ShaderNodeValToRGB")
    emit = nodes.new("ShaderNodeEmission")
    base = rgba(value)
    diff.inputs["Color"].default_value = base
    diff.inputs["Roughness"].default_value = 0.82
    cr = ramp.color_ramp
    cr.interpolation = "CONSTANT"
    cr.elements[0].position = 0.34
    cr.elements[0].color = shade(base, 0.38)
    cr.elements[1].position = 0.72
    cr.elements[1].color = base
    mid = cr.elements.new(0.54)
    mid.color = shade(base, 0.67)
    emit.inputs["Strength"].default_value = 1.0
    links.new(diff.outputs["BSDF"], to_rgb.inputs["Shader"])
    links.new(to_rgb.outputs["Color"], ramp.inputs["Fac"])
    links.new(ramp.outputs["Color"], emit.inputs["Color"])
    links.new(emit.outputs["Emission"], out.inputs["Surface"])
    return mat


def reset() -> None:
    bpy.ops.object.select_all(action="SELECT")
    bpy.ops.object.delete(use_global=False)
    for collection in tuple(bpy.data.collections):
        if collection.name != "Collection":
            bpy.data.collections.remove(collection)


def collection(name: str) -> bpy.types.Collection:
    c = bpy.data.collections.new(name)
    bpy.context.scene.collection.children.link(c)
    return c


def relink(obj: bpy.types.Object, c: bpy.types.Collection) -> None:
    for old in tuple(obj.users_collection):
        old.objects.unlink(obj)
    c.objects.link(obj)


def group(name: str, root: bpy.types.Object, c: bpy.types.Collection) -> bpy.types.Object:
    obj = bpy.data.objects.new(name, None)
    obj.parent = root
    c.objects.link(obj)
    return obj


def harden(obj: bpy.types.Object, width: float = 0.045) -> bpy.types.Object:
    bevel = obj.modifiers.new("HAND_AUTHORED_EDGE_BREAK", "BEVEL")
    bevel.width = width
    bevel.segments = 2
    bevel.affect = "EDGES"
    return obj


def box(name, size, loc, mat, parent, c, rot=(0, 0, 0), bevel=0.06):
    bpy.ops.mesh.primitive_cube_add(location=loc, rotation=rot)
    obj = bpy.context.object
    obj.name = name
    obj.scale = tuple(v / 2 for v in size)
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    obj.data.materials.append(mat)
    obj.parent = parent
    relink(obj, c)
    return harden(obj, min(bevel, min(size) * 0.16))


def prism(name, points_xz, y_front, y_back, mat, parent, c, bevel=0.04):
    n = len(points_xz)
    verts = [(x, y_front, z) for x, z in points_xz] + [(x, y_back, z) for x, z in points_xz]
    faces = [tuple(range(n - 1, -1, -1)), tuple(range(n, n * 2))]
    for i in range(n):
        j = (i + 1) % n
        faces.append((i, n + i, n + j, j))
    mesh = bpy.data.meshes.new(name + "_MESH")
    mesh.from_pydata(verts, [], faces)
    mesh.update()
    obj = bpy.data.objects.new(name, mesh)
    obj.data.materials.append(mat)
    obj.parent = parent
    c.objects.link(obj)
    return harden(obj, bevel)


def cyl(name, radius, depth, loc, mat, parent, c, rot=(0, 0, 0), vertices=24, bevel=0.035):
    bpy.ops.mesh.primitive_cylinder_add(vertices=vertices, radius=radius, depth=depth, location=loc, rotation=rot)
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(mat)
    obj.parent = parent
    relink(obj, c)
    return harden(obj, bevel)


def torus(name, major, minor, loc, mat, parent, c, rot=(0, 0, 0)):
    bpy.ops.mesh.primitive_torus_add(major_radius=major, minor_radius=minor, major_segments=32, minor_segments=8, location=loc, rotation=rot)
    obj = bpy.context.object
    obj.name = name
    obj.data.materials.append(mat)
    obj.parent = parent
    relink(obj, c)
    return obj


def ellipsoid(name, loc, scale, mat, parent, c, rot=(0, 0, 0), subdivisions=3):
    bpy.ops.mesh.primitive_ico_sphere_add(subdivisions=subdivisions, radius=1, location=loc, rotation=rot)
    obj = bpy.context.object
    obj.name = name
    obj.scale = scale
    bpy.ops.object.transform_apply(location=False, rotation=False, scale=True)
    obj.data.materials.append(mat)
    obj.parent = parent
    relink(obj, c)
    return obj


def beam(name, start, end, width, depth, mat, parent, c, bevel=0.04):
    a, b = Vector(start), Vector(end)
    d = b - a
    obj = box(name, (width, depth, d.length), (a + b) / 2, mat, parent, c, bevel=bevel)
    obj.rotation_mode = "QUATERNION"
    obj.rotation_quaternion = Vector((0, 0, 1)).rotation_difference(d.normalized())
    return obj


def taper(name, start, end, r0, r1, mat, parent, c, vertices=8):
    a, b = Vector(start), Vector(end)
    d = b - a
    bpy.ops.mesh.primitive_cone_add(vertices=vertices, radius1=r1, radius2=r0, depth=d.length, location=(a + b) / 2)
    obj = bpy.context.object
    obj.name = name
    obj.rotation_mode = "QUATERNION"
    obj.rotation_quaternion = Vector((0, 0, 1)).rotation_difference(d.normalized())
    obj.data.materials.append(mat)
    obj.parent = parent
    relink(obj, c)
    return harden(obj, 0.035)

        beam(f"flex_seam_{i}", a, b, w, .14, m["bx_seam"], g["SEAMS"], c, .018)
    beam("visible_shell_suture_L", (-2.28, -2.02, 4.92), (-.78, -2.18, 5.36), .085, .12, m["bx0"], g["SEAMS"], c, .014)
    beam("visible_shell_suture_R", (.12, -2.16, 5.46), (1.62, -1.94, 5.08), .07, .1, m["bx_seam"], g["SEAMS"], c, .014)
    ellipsoid("left_membrane", (-1.88, -.5, 4.2), (.5, 1.08, .78), m["bx0"], g["SEAMS"], c, rot=(.12, .16, -.24), subdivisions=2)
    ellipsoid("right_membrane", (1.78, -.04, 4.08), (.42, .96, .68), m["bx1"], g["SEAMS"], c, rot=(.1, -.14, .2), subdivisions=2)

    # Low sensor head is embedded under the shell lip.
    prism("head_wedge", ((-1.04, 3.74), (1.0, 3.68), (.76, 3.28), (-.9, 3.24)), -3.08, -2.2, m["bx1"], g["HEAD"], c, .08)
    prism("head_brow_lip", ((-1.32, 3.88), (1.18, 3.8), (.98, 3.6), (-1.16, 3.64)), -3.2, -2.5, m["bx3"], g["HEAD"], c, .045)
    box("single_red_slit", (.68, .1, .06), (-.1, -3.28, 3.5), m["bx_red"], g["HEAD"], c, rot=(0, 0, -.04), bevel=.012)
    prism("closed_chest_L", ((-1.1, 3.3), (-.06, 3.38), (-.08, 2.62), (-.76, 2.46)), -2.36, -1.74, m["bx2"], g["SHELL"], c, .075)
    prism("closed_chest_R", ((.08, 3.38), (.88, 3.24), (.66, 2.5), (.08, 2.62)), -2.32, -1.76, m["bx3"], g["SHELL"], c, .075)
    box("closed_chest_seam", (.055, .1, .76), (0, -2.46, 2.94), m["bx0"], g["SEAMS"], c, rot=(-.05, 0, 0), bevel=.012)
    prism("asymmetric_ventral_shield", ((-.92, 3.2), (.34, 3.28), (.52, 2.52), (-.66, 2.34)), -2.48, -2.3, m["bx1"], g["SHELL"], c, .05)

    # Two unequal siege pylons carry the mass; pads are anchors rather than paws.
    specs = {
        "L": ((-2.34, -.72, 4.08), (-2.5, -1.5, 2.3), (-2.54, -2.34, .4), .72, .6, "bx3"),
        "R": ((1.98, -.42, 3.94), (2.16, -1.24, 2.08), (2.2, -2.04, .38), .62, .52, "bx2"),
    }
    for s, (shoulder, hinge, contact, r0, r1, mat) in specs.items():
        grp = g[f"FORE_{s}"]
        ellipsoid(f"load_socket_{s}", shoulder, (r0, r0 * .78, r0 * .7), m["bx2" if s == "L" else "bx1"], grp, c, rot=(.2, .12 if s == "L" else -.1, .18), subdivisions=2)
        beam(f"root_tendon_{s}", (shoulder[0] - (.34 if s == "L" else -.26), shoulder[1] - .5, shoulder[2] + .14), (hinge[0], hinge[1] - .38, hinge[2] + .28), .08 if s == "L" else .06, .12, m["bx_seam"], grp, c, .014)
        taper(f"upper_pylon_{s}", shoulder, hinge, r0, r1, m[mat], grp, c, 7 if s == "L" else 6)
        ellipsoid(f"flex_hinge_{s}", hinge, (r1 * 1.05, r1, r1 * .82), m["bx0"], grp, c, rot=(.18, .1, .2), subdivisions=2)
        taper(f"lower_pylon_{s}", hinge, contact, r1, r1 * 1.1, m["bx1"], grp, c, 6 if s == "L" else 7)
        ellipsoid(f"pylon_overplate_{s}", tuple((shoulder[i] + hinge[i]) / 2 for i in range(3)), (r0 * .78, r0 * .5, .94), m["bx4" if s == "L" else "bx2"], grp, c, rot=(.2, .12 if s == "L" else -.12, .18 if s == "L" else -.12), subdivisions=2)
        prism(f"anchor_pad_{s}", ((contact[0] - .7, .08), (contact[0] + .64, .08), (contact[0] + .72, .3), (contact[0] + .22, .52), (contact[0] - .58, .44)), contact[1] - .72, contact[1] + .34, m["bx2"], grp, c, .055)
        beam(f"anchor_flex_{s}", (contact[0] - .42, contact[1] - .62, .18), (contact[0] + .38, contact[1] - .7, .14), .14, .24, m["bx0"], grp, c, .03)

    # Short rear stabilizers remain subordinate and partially hidden by the shell.
    for sign, s in ((-1, "L"), (1, "R")):
        grp = g[f"REAR_{s}"]
        a = (sign * (.7 if s == "L" else .58), 1.72 if s == "L" else 1.94, .52)
        b = (sign * (.74 if s == "L" else .62), 1.96 if s == "L" else 2.16, .2)
        ellipsoid(f"rear_socket_{s}", a, (.22, .28, .18), m["bx3"], grp, c, subdivisions=2)
        taper(f"rear_stabilizer_{s}", a, b, .2, .28, m["bx1"], grp, c, 6)
        ellipsoid(f"rear_pad_{s}", (b[0], b[1] + .04, .1), (.3, .28, .09), m["bx2"], grp, c, subdivisions=2)
    ellipsoid("sealed_exit_shell", (.2, 2.36, 4.44), (.62, .28, .4), m["bx1"], g["SEALED_EXIT"], c, rot=(.1, .06, .04), subdivisions=2)
    ellipsoid("sealed_exit_cap", (.2, 2.58, 4.44), (.36, .12, .22), m["bx0"], g["SEALED_EXIT"], c, rot=(.1, .06, .04), subdivisions=2)
    return c, root


def scene_setup():
    scene = bpy.context.scene
    scene.render.engine = "BLENDER_EEVEE_NEXT"
    scene.render.resolution_x = 1024
    scene.render.resolution_y = 1024
    scene.render.resolution_percentage = 100
    scene.render.image_settings.file_format = "PNG"
    scene.render.image_settings.color_mode = "RGBA"
    scene.render.image_settings.color_depth = "8"
    scene.render.film_transparent = False
    scene.render.use_freestyle = True
    scene.render.line_thickness = 2.1
    scene.view_settings.view_transform = "Standard"
    scene.view_settings.look = "Medium High Contrast"
    scene.world.use_nodes = True
    world_bg = scene.world.node_tree.nodes.get("Background")
    world_bg.inputs["Color"].default_value = rgba("e9ece8")
    world_bg.inputs["Strength"].default_value = 0.8
    lines = bpy.context.view_layer.freestyle_settings.linesets[0].linestyle
    lines.color = rgba(PALETTE["ink"])[:3]
    lines.thickness = 2.2
    bpy.context.view_layer.freestyle_settings.linesets[0].select_silhouette = True
    bpy.context.view_layer.freestyle_settings.linesets[0].select_crease = True
    bpy.context.view_layer.freestyle_settings.linesets[0].select_border = True
    for kind, loc, energy, size, col in (
        ("AREA", (-7, -10, 14), 1250, 6, (1.0, .94, .84)),
        ("AREA", (8, -4, 9), 320, 5, (.5, .72, .86)),
        ("AREA", (0, 8, 12), 700, 5, (.9, .48, .28)),
    ):
        bpy.ops.object.light_add(type=kind, location=loc)
        light = bpy.context.object
        light.data.energy = energy
        light.data.shape = "DISK"
        light.data.size = size
        light.data.color = col
    bpy.ops.object.camera_add(location=(0, -20, 4.1))
    cam = bpy.context.object
    cam.data.type = "ORTHO"
    scene.camera = cam
    return cam


def aim(cam, target):
    cam.rotation_euler = (Vector(target) - cam.location).to_track_quat("-Z", "Y").to_euler()


def render(c, all_c, cam, loc, target, scale, filename):
    for item in all_c:
        item.hide_render = item != c
        item.hide_viewport = item != c
    cam.location = loc
    cam.data.ortho_scale = scale
    aim(cam, target)
    bpy.context.scene.render.filepath = str(OUT / filename)
    bpy.ops.render.render(write_still=True)


def main():
    reset()
    mats = {name: cel_mat("HD_" + name.upper(), value) for name, value in PALETTE.items()}
    gz, _ = guardian(mats)
    bx, _ = breaker(mats)
    cam = scene_setup()
    all_c = (gz, bx)

    bpy.ops.wm.save_as_mainfile(filepath=str(HERE / "work/blender/A91A3_HD_HAND_AUTHORED_GATE.blend"))
