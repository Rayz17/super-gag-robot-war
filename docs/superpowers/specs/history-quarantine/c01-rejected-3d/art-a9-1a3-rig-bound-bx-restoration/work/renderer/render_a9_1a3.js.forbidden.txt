const toonCache = new Map();
function toon(color) {
  if (!toonCache.has(color)) {
    toonCache.set(color, new THREE.MeshToonMaterial({ color, gradientMap: gradient }));
  }
  return toonCache.get(color);
}

const inkMaterial = new THREE.MeshBasicMaterial({ color: C.ink, side: THREE.BackSide });
const edgeMaterial = new THREE.LineBasicMaterial({ color: C.ink, transparent: true, opacity: 0.96 });

function componentGroup(root, name) {
  const group = new THREE.Group();
  group.name = name;
  group.userData.component = name;
  root.add(group);
  return group;
}

function finish(mesh, component, edgeThreshold = 26, outlineScale = 1.024) {
  mesh.userData.primary = true;
  mesh.userData.component = component;
  mesh.userData.baseColor = mesh.material.color.getHex();
  const outline = new THREE.Mesh(mesh.geometry, inkMaterial);
  outline.userData.isOutline = true;
  outline.scale.setScalar(outlineScale);
  outline.renderOrder = -3;
  mesh.add(outline);
  const edges = new THREE.LineSegments(new THREE.EdgesGeometry(mesh.geometry, edgeThreshold), edgeMaterial);
  edges.userData.isEdge = true;
  edges.scale.setScalar(1.003);
  edges.renderOrder = 4;
  mesh.add(edges);
  return mesh;
}

function box(parent, name, size, position, color, component, rotation = [0, 0, 0], outlineScale = 1.024) {
  const mesh = finish(new THREE.Mesh(new THREE.BoxGeometry(...size), toon(color)), component, 22, outlineScale);
  mesh.name = name;
  mesh.position.set(...position);
  mesh.rotation.set(...rotation);
  parent.add(mesh);
  return mesh;
}

function chamferBox(parent, name, size, position, color, component, rotation = [0, 0, 0], chamfer = 0.12) {
  const [width, height, depth] = size;
  const cut = Math.min(chamfer, width * 0.22, height * 0.22);
  const shape = new THREE.Shape();
  const points = [
    [-width / 2 + cut, -height / 2], [width / 2 - cut, -height / 2],
    [width / 2, -height / 2 + cut], [width / 2, height / 2 - cut],
    [width / 2 - cut, height / 2], [-width / 2 + cut, height / 2],
    [-width / 2, height / 2 - cut], [-width / 2, -height / 2 + cut],
  ];
  shape.moveTo(points[0][0], points[0][1]);
  for (let i = 1; i < points.length; i += 1) shape.lineTo(points[i][0], points[i][1]);
  shape.closePath();
  const bevel = Math.min(0.07, depth * 0.1, cut * 0.45);
  const geometry = new THREE.ExtrudeGeometry(shape, {
    depth: Math.max(0.02, depth - bevel * 2), bevelEnabled: true,
    bevelSegments: 1, bevelSize: bevel, bevelThickness: bevel, steps: 1,
  });
  geometry.translate(0, 0, -depth / 2 + bevel);
  const mesh = finish(new THREE.Mesh(geometry, toon(color)), component, 16, 1.022);
  mesh.name = name;
  mesh.position.set(...position);
  mesh.rotation.set(...rotation);
  parent.add(mesh);
  return mesh;
}

function cylinder(parent, name, radiusTop, radiusBottom, height, sides, position, color, component, rotation = [0, 0, 0], outlineScale = 1.024) {
  const geometry = new THREE.CylinderGeometry(radiusTop, radiusBottom, height, sides, 1, false);
  const mesh = finish(new THREE.Mesh(geometry, toon(color)), component, 18, outlineScale);
  mesh.name = name;
  mesh.position.set(...position);
  mesh.rotation.set(...rotation);
  parent.add(mesh);
  return mesh;
}

function poly(parent, name, radius, detail, position, scale, color, component, rotation = [0, 0, 0]) {
  const mesh = finish(new THREE.Mesh(new THREE.DodecahedronGeometry(radius, detail), toon(color)), component, 15, 1.035);
  mesh.name = name;
  mesh.position.set(...position);
  mesh.scale.set(...scale);
  mesh.rotation.set(...rotation);
  parent.add(mesh);
  return mesh;
}

function beam(parent, name, start, end, width, depth, color, component, outlineScale = 1.03) {
  const a = new THREE.Vector3(...start);
  const b = new THREE.Vector3(...end);
  const direction = b.clone().sub(a);
  const mesh = finish(new THREE.Mesh(new THREE.BoxGeometry(width, direction.length(), depth), toon(color)), component, 20, outlineScale);
  mesh.name = name;
  mesh.position.copy(a.clone().add(b).multiplyScalar(0.5));
  mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), direction.clone().normalize());
  parent.add(mesh);
  return mesh;
}

function plateXY(parent, name, points, depth, z, color, component) {
  const shape = new THREE.Shape();
  shape.moveTo(points[0][0], points[0][1]);
  for (let i = 1; i < points.length; i += 1) shape.lineTo(points[i][0], points[i][1]);
  shape.closePath();
  const geometry = new THREE.ExtrudeGeometry(shape, { depth, bevelEnabled: false, steps: 1 });
  geometry.translate(0, 0, -depth / 2);
  const mesh = finish(new THREE.Mesh(geometry, toon(color)), component, 18, 1.025);
  mesh.name = name;
  mesh.position.z = z;
  parent.add(mesh);
  return mesh;
}

function applyRenderMode(root) {
  const componentColors = {};
  for (const [name, color] of Object.entries(MASK)) componentColors[name] = `#${color.toString(16).padStart(6, '0')}`;
  root.traverse((object) => {
    if (object.userData.isEdge) {
      object.visible = mode === 'color';
      return;
    }
    if (object.userData.isOutline) {
      object.visible = mode !== 'components';
      if (mode === 'silhouette') object.material = new THREE.MeshBasicMaterial({ color: 0xffffff, side: THREE.BackSide });
      return;
    }
    if (!object.userData.primary) return;
    if (mode === 'silhouette') object.material = new THREE.MeshBasicMaterial({ color: 0xffffff });
    if (mode === 'components') object.material = new THREE.MeshBasicMaterial({ color: MASK[object.userData.component] || 0xffffff });
  });
  return componentColors;
}

function configureCamera(model) {
  const isGz = asset === 'guardian_z';
  const aspect = window.innerWidth / window.innerHeight;
  const height = isGz ? 9.75 : 9.5;
  camera.left = (-height * aspect) / 2;
  camera.right = (height * aspect) / 2;
  camera.top = height / 2;
  camera.bottom = -height / 2;
  camera.near = 0.1;
  camera.far = 100;
  const targetY = isGz ? 4.0 : 2.9;
  if (isGz) {
    if (view === 'front') camera.position.set(0, targetY, 18);
    else if (view === 'back') camera.position.set(0, targetY, -18);
    else if (view === 'left') camera.position.set(-18, targetY, 0);
    else camera.position.set(18, targetY, 0);
  } else {
    camera.position.set(view === 'contact' ? 10.5 : 9.2, 7.1, view === 'contact' ? 14.2 : 15.5);
  }
  camera.lookAt(0, targetY, 0);
  camera.updateProjectionMatrix();
  model.updateMatrixWorld(true);
}

function projectedKeypoints(model) {
  const result = {};
  const width = window.innerWidth;
  const height = window.innerHeight;
  for (const [name, coords] of Object.entries(model.userData.keypoints)) {
    const point = new THREE.Vector3(...coords).applyMatrix4(model.matrixWorld).project(camera);
    result[name] = {
      x: Number((((point.x + 1) / 2) * width).toFixed(3)),
      y: Number((((1 - point.y) / 2) * height).toFixed(3)),
      ndc_z: Number(point.z.toFixed(5)),
    };
  }
  return result;
}
