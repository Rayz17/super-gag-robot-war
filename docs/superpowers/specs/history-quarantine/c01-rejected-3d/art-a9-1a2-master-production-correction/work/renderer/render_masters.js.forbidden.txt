const toonMaterials = new Map();
function toon(color) {
  if (!toonMaterials.has(color)) {
    toonMaterials.set(color, new THREE.MeshToonMaterial({ color, gradientMap: gradient }));
  }
  return toonMaterials.get(color);
}

const outlineMaterial = new THREE.MeshBasicMaterial({ color: COLORS.ink, side: THREE.BackSide });
const edgeMaterial = new THREE.LineBasicMaterial({ color: COLORS.ink, transparent: true, opacity: 0.92 });

function finishMesh(mesh, edgeThreshold = 28, outlineScale = 1.026) {
  const outline = new THREE.Mesh(mesh.geometry, outlineMaterial);
  outline.scale.setScalar(outlineScale);
  outline.renderOrder = -2;
  mesh.add(outline);

  const edges = new THREE.LineSegments(new THREE.EdgesGeometry(mesh.geometry, edgeThreshold), edgeMaterial);
  edges.scale.setScalar(1.004);
  edges.renderOrder = 3;
  mesh.add(edges);
  return mesh;
}

function addBox(parent, name, size, position, color, rotation = [0, 0, 0], outlineScale = 1.026) {
  const mesh = finishMesh(new THREE.Mesh(new THREE.BoxGeometry(...size), toon(color)), 24, outlineScale);
  mesh.name = name;
  mesh.position.set(...position);
  mesh.rotation.set(...rotation);
  parent.add(mesh);
  return mesh;
}

function addCylinder(parent, name, radiusTop, radiusBottom, height, radialSegments, position, color, rotation = [0, 0, 0], outlineScale = 1.026) {
  const mesh = finishMesh(
    new THREE.Mesh(new THREE.CylinderGeometry(radiusTop, radiusBottom, height, radialSegments, 1, false), toon(color)),
    20,
    outlineScale,
  );
  mesh.name = name;
  mesh.position.set(...position);
  mesh.rotation.set(...rotation);
  parent.add(mesh);
  return mesh;
}

function addBeam(parent, name, start, end, width, depth, color) {
  const a = new THREE.Vector3(...start);
  const b = new THREE.Vector3(...end);
  const direction = b.clone().sub(a);
  const mesh = finishMesh(new THREE.Mesh(new THREE.BoxGeometry(width, direction.length(), depth), toon(color)), 22, 1.035);
  mesh.name = name;
  mesh.position.copy(a.clone().add(b).multiplyScalar(0.5));
  mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), direction.clone().normalize());
  parent.add(mesh);
  return mesh;
}

function addFrontDisc(parent, name, radius, depth, position, color, segments = 16) {
  return addCylinder(parent, name, radius, radius, depth, segments, position, color, [Math.PI / 2, 0, 0], 1.03);
}

function configureCamera(kind, model) {
  const aspect = window.innerWidth / window.innerHeight;
  const isGuardian = kind.startsWith('gz_');
  const orthoHeight = isGuardian ? 9.25 : 8.3;
  camera.left = (-orthoHeight * aspect) / 2;
  camera.right = (orthoHeight * aspect) / 2;
  camera.top = orthoHeight / 2;
  camera.bottom = -orthoHeight / 2;
  camera.near = 0.1;
  camera.far = 100;
  camera.updateProjectionMatrix();

  const targetY = isGuardian ? 4.05 : 2.45;
  const view = kind.split('_').at(-1);
  if (view === 'front' || view === 'planning' || view === 'contact') camera.position.set(0, targetY, 18);
  else if (view === 'back') camera.position.set(0, targetY, -18);
  else if (view === 'left') camera.position.set(-18, targetY, 0);
  else if (view === 'right') camera.position.set(18, targetY, 0);
  else if (view === 'iso') camera.position.set(11, isGuardian ? 9.5 : 7.5, 14);
  else camera.position.set(0, targetY, 18);
  camera.lookAt(0, targetY, 0);
  model.updateMatrixWorld(true);
}

if (asset.startsWith('gz_')) model = buildGuardianZ();
else model = buildBreaker(asset.endsWith('contact') ? 'contact' : 'planning');
scene.add(model);
configureCamera(asset, model);
renderer.render(scene, camera);
